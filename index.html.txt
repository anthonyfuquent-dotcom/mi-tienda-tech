        // Catálogo por defecto
        const productosIniciales = [

            },
            {
                id: 2,
                nombre: "Mouse Gamer Ergónomico",
                precio: 28.00,
                imagen: "https://images.unsplash.com/photo-1615663245857-ac93bb7c39e7?auto=format&fit=crop&w=600&q=80",
                desc: "Sensor óptico de 12,000 DPI, agarre antideslizante y botones laterales programables."
            },
            {
                id: 3,
                nombre: "Audífonos Surround 7.1",
                precio: 52.00,
                imagen: "https://images.unsplash.com/photo-1546435770-a3e426bf472b?auto=format&fit=crop&w=600&q=80",
                desc: "Audio posicional 7.1, micrófonos con cancelación de ruido y almohadillas confortables."
            }
        ];

        let modoAdmin = false;

        function obtenerProductos() {
            const guardados = localStorage.getItem('premiumzone_prods');
            return guardados ? JSON.parse(guardados) : productosIniciales;
        }

        function renderizarProductos() {
            const productos = obtenerProductos();
            const contenedor = document.getElementById('contenedorProductos');
            contenedor.innerHTML = '';

            productos.forEach(p => {
                contenedor.innerHTML += `
                    <div class="card">
                        <div class="card-img-wrapper">
                            <img src="${p.imagen}" alt="${p.nombre}" onerror="this.src='https://via.placeholder.com/300x190?text=Sin+Imagen'">
                        </div>
                        <div class="card-body">
                            <div>
                                <h3 class="card-title">${p.nombre}</h3>
                                <p class="card-desc">${p.desc}</p>
                            </div>
                            <div>
                                <div class="card-price">$${parseFloat(p.precio).toFixed(2)}</div>
                                <button class="btn-ws" onclick="comprarProducto('${p.nombre}')">Comprar por WhatsApp</button>
                                ${modoAdmin ? `
                                    <button class="btn-edit" onclick="editarProducto(${p.id})">✏️ Editar</button>
                                    <button class="btn-del" onclick="eliminarProducto(${p.id})">🗑️ Eliminar</button>
                                ` : ''}
                            </div>
                        </div>
                    </div>
                `;
            });
        }

        function toggleAdmin() {
            modoAdmin = !modoAdmin;
            document.getElementById('adminPanel').style.display = modoAdmin ? 'block' : 'none';
            renderizarProductos();
        }

        function guardarProducto(e) {
            e.preventDefault();
            const id = document.getElementById('prodId').value;
            const nombre = document.getElementById('prodNombre').value;
            const precio = document.getElementById('prodPrecio').value;
            const imagen = document.getElementById('prodImagen').value;
            const desc = document.getElementById('prodDesc').value;

            let productos = obtenerProductos();

            if (id) {
                productos = productos.map(p => p.id == id ? { id: Number(id), nombre, precio, imagen, desc } : p);
            } else {
                productos.push({ id: Date.now(), nombre, precio, imagen, desc });
            }

            localStorage.setItem('premiumzone_prods', JSON.stringify(productos));
            document.getElementById('productForm').reset();
            document.getElementById('prodId').value = '';
            renderizarProductos();
        }

        function editarProducto(id) {
            const productos = obtenerProductos();
            const p = productos.find(item => item.id === id);
            if (p) {
                document.getElementById('prodId').value = p.id;
                document.getElementById('prodNombre').value = p.nombre;
                document.getElementById('prodPrecio').value = p.precio;
                document.getElementById('prodImagen').value = p.imagen;
                document.getElementById('prodDesc').value = p.desc;
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }
        }

        function eliminarProducto(id) {
            if (confirm('¿Deseas eliminar este producto del catálogo?')) {
                let productos = obtenerProductos();
                productos = productos.filter(p => p.id !== id);
                localStorage.setItem('premiumzone_prods', JSON.stringify(productos));
                renderizarProductos();
            }
        }

        function comprarProducto(producto) {
            const msg = encodeURIComponent(`Hola Premium Zone, estoy interesado en comprar: ${producto}`);
            window.open(`https://wa.me/${TEL_WHATSAPP}?text=${msg}`, '_blank');
        }

        function solicitarServicio(e) {
            e.preventDefault();
            const tipo = document.getElementById('tipoServicio').value;
            const detalles = document.getElementById('detallesServicio').value;
            const msg = encodeURIComponent(`Hola Premium Zone, quisiera realizar una consulta:\n\n*Servicio:* ${tipo}\n*Detalles:* ${detalles}`);
            window.open(`https://wa.me/${TEL_WHATSAPP}?text=${msg}`, '_blank');
        }

        function contactarSoporte() {
            const msg = encodeURIComponent("Hola Premium Zone, requiero atención al cliente para realizar una consulta sobre garantías o soporte.");
            window.open(`https://wa.me/${TEL_WHATSAPP}?text=${msg}`, '_blank');
        }

        // Cargar catálogo al abrir
        renderizarProductos();
    </script>
